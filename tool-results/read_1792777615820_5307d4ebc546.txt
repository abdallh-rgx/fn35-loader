    79→def sysctl_bytes(name):
    80→    v = SYSCTL_BY_NAME.get(name)
    81→    if v is None:
    82→        return None
    83→    if isinstance(v, int):
    84→        if 0 <= v < 0x100000000:
    85→            return struct.pack("<I", v)
    86→        return struct.pack("<q", v)
    87→    return v

   561→@cfunc("__tlv_get_addr", "_tlv_bootstrap")
   562→def _tlv_get_addr(l, desc):
   563→    key = l.rd64(desc + 8)
   564→    if key == 0:
   565→        key = desc
   566→        l.w64(desc + 8, key)
   567→    offset = l.rd64(desc + 16)
   568→    t = l.cur
   569→    base = t.tlv_bases.get(key)
   570→    if base is None:
   571→        tlv_base, tlv_size = l.tlv_template
   572→        base = l.heap_alloc(max(tlv_size, 16), zero=True)
   573→        if tlv_base and tlv_size:
   574→            try:

   856→# ====================================================================
   857→# native-callback qsort
   858→# ====================================================================
   859→
   860→def _qsort_native(l, base, n, size, compar):
   861→    def drv():
   862→        arr = [base + i * size for i in range(_to_int(n))]
   863→        for i in range(1, len(arr)):
   864→            j = i
   865→            while j > 0:
   866→                r = yield ("call", compar, (arr[j - 1], arr[j]))
   867→                if r <= 0:
   868→                    break
   869→                a1 = l.rdb(arr[j - 1], size)
   870→                a2 = l.rdb(arr[j], size)
   871→                l.wrb(arr[j - 1], a2)
   872→                l.wrb(arr[j], a1)
   873→                arr[j - 1], arr[j] = base + (j - 1) * size, base + j * size
   874→                j -= 1
   875→
   876→    l.py_call(drv(), ret=0)
   877→    return 0

   889→@cfunc("_host_statistics", "_host_statistics64", "_host_processor_info")
   890→def _host_statistics(l, host, flavor, info, count):
   891→    # flavor 2 = HOST_BASIC_INFO; fill a real struct (drives UE4's
   892→    # memory model and page-size computations)
   893→    try:
   894→        if flavor == 2 and info:
   895→            # NOTE: UE4's FMallocBinned2 pool sizing derives from
   896→            # max_cpus * PageSize; needs >= 8 MB to build a real hash
   897→            # table. Core counts used by the engine itself come from
   898→            # sysctl(hw.ncpu) instead, so this stays a boot parameter.
   899→            data = struct.pack(
   900→                "<iiIiiiiiiI Q",
   901→                512,            # max_cpus (pool sizing)
   902→                512,            # avail_cpus

   904→                0x0100000C,     # cpu_type (ARM64)
   905→                0,              # cpu_subtype
   906→                0,              # thread_toggle
   907→                6,              # physical_cpu
   908→                6,              # physical_cpu_max
   909→                6,              # logical_cpu
   910→                6,              # logical_cpu_max
   911→                3221225472)     # max_mem
   912→            l.wrb(info, data)

   915→        elif flavor == 1 and info:  # HOST_LOAD_INFO

   917→            l.wrb(info, data)
